11.1 为什么亲子家庭领域必须在本地跑？

亲子教育、青少年叛逆、夫妻教育分歧等家庭心理咨询，具有两个极其鲜明的行业特性：

  1. 隐私敏感度极高（绝不愿上公有云）：
    没有家长愿意把自己和孩子最私密的冲突、孩子的真实姓名、敏感的情绪焦虑上传到商业闭源 API（如 ChatGPT、百度等）。一旦数据被云端记录或用作二次训练，存在极大的心理负担。本地纯离线运行，网线拔掉依然能聊，是家庭顾问场景的唯一安全解法。

▶ 单元格 2 · 亲子数据微调、Q4_K_M 量化与 Google Drive 归档（约 25 分钟：GPU 微调 7 分钟 + 原厂权重下载 6 分钟 + CPU 逐层量化 12 分钟）

import os, gc, torch, glob, shutil
from datasets import load_dataset
from unsloth import FastLanguageModel, train_on_responses_only
from trl import SFTTrainer
from transformers import TrainingArguments
from google.colab import drive

# 1. 加载最新一代 Qwen3-4B 黄金甜点底座与 LoRA 适配器
print("1. 正在载入最新一代 Qwen3-4B 底座模型 ...")
model, tokenizer = FastLanguageModel.from_pretrained(
    model_name = "unsloth/Qwen3-4B-unsloth-bnb-4bit",
    max_seq_length = 2048,
    load_in_4bit = True,
)

model = FastLanguageModel.get_peft_model(
    model,
    r = 16,
    target_modules = ["q_proj", "k_proj", "v_proj", "o_proj", "gate_proj", "up_proj", "down_proj"],
    lora_alpha = 16,
    lora_dropout = 0,
    bias = "none",
    use_gradient_checkpointing = "unsloth",
)

# 2. 载入中科院心理所 CPsyCoun 咨询数据集，并精准筛选亲子家庭记录
print("2. 正在拉取中科院亲子心理与家庭咨询数据集 ...")
raw_data = load_dataset("CAS-SIAT-XinHai/CPsyCoun", split="train")

parenting_keywords = ["孩子", "儿子", "女儿", "父母", "妈妈", "爸爸", "家庭", "叛逆", "上学", "沟通", "发脾气", "吵架"]

def is_parenting_record(row):
    full_text = row.get("instruction", "") + " " + row.get("output", "")
    for pair in row.get("history", []):
        full_text += " " + pair[0] + " " + pair[1]
    return any(kw in full_text for kw in parenting_keywords)

parenting_subset = raw_data.filter(is_parenting_record)
print(f"筛选完成！共挖掘出 {len(parenting_subset)} 条亲子家庭深度多轮问答")

# 3. 格式转化：转换为带系统人设的 Qwen3-4B ChatML 格式
def convert_to_chatml(row):
    msgs = [{
        "role": "system",
        "content": "你是一位温暖、富有同理心的中国亲子教育与家庭心理顾问。请运用非暴力沟通技巧，先接纳家长的焦虑情绪，再从儿童心理发展规律出发，给出温柔、坚定、具体可行的家庭沟通建议。"
    }]
    for q, a in row.get("history", []):
        msgs.append({"role": "user", "content": q.strip()})
        msgs.append({"role": "assistant", "content": a.strip()})
    
    current_q = (row.get("instruction", "") + " " + row.get("input", "")).strip()
    if current_q:
        msgs.append({"role": "user", "content": current_q})
    if row.get("output", ""):
        msgs.append({"role": "assistant", "content": row["output"].strip()})
    
    formatted = tokenizer.apply_chat_template(msgs, tokenize=False, add_generation_prompt=False)
    return {"text": formatted}

# 抽选 200 条最精华数据进行训练
train_pool = parenting_subset.shuffle(seed=42).select(range(min(200, len(parenting_subset))))
dataset = train_pool.map(convert_to_chatml)

# 4. 配置训练参数并启用答案遮罩（4B 显存优化配置：批次2 + 累加4）
args = TrainingArguments(
    output_dir = "outputs_parenting_qwen3",
    per_device_train_batch_size = 2,     # 物理小批次设为 2（防 OOM）
    gradient_accumulation_steps = 4,     # 梯度累加 4 次 ➔ 有效批次依然为 2×4=8！
    num_train_epochs = 2,
    learning_rate = 2e-4,
    logging_steps = 10,
    save_strategy = "no",
    report_to = "none"
)

trainer = SFTTrainer(
    model = model,
    tokenizer = tokenizer,
    train_dataset = dataset,
    dataset_text_field = "text",
    max_seq_length = 2048,
    args = args,
)
trainer = train_on_responses_only(
    trainer,
    instruction_part = "<|im_start|>user\n",
    response_part = "<|im_start|>assistant\n",
)

trainer.model.config.use_cache = False
print("3. 开始微调 Qwen3-4B 亲子心理专家模型（约 3 分钟）...")
trainer.train()

# 5. 物理焊死踏板并直接导出为 Q4_K_M GGUF 单文件
export_dir = "parenting_qwen3_gguf"
print("4. 正在合并 LoRA 并导出为 4-bit GGUF 量化文件 ...")
model.save_pretrained_gguf(
    export_dir,
    tokenizer,
    quantization_method = "q4_k_m"
)

# 6. 幂等检查并挂载 Google Drive（已挂载则跳过，未挂载才唤起授权）
print("5. 检查并挂载 Google Drive 云端硬盘 ...")
if not os.path.exists('/content/drive/MyDrive'):

    drive.mount('/content/drive')
else:

drive_save_dir = "/content/drive/MyDrive/ai-models/parenting-qwen3-4b"
os.makedirs(drive_save_dir, exist_ok=True)

gguf_candidates = glob.glob(f"{export_dir}*/*.gguf") + glob.glob(f"{export_dir}*/*/*.gguf")
if gguf_candidates:
    target_file = gguf_candidates[0]

硬核机理：为什么不能直接把训练时的 4-bit 模型导出？

很多初学者会大吃一惊：“我们训练时不就是 4-bit 吗？为什么第 4 步导出时还要去下载 8GB 的原始权重？”

答案在于矩阵加法的精度不可逆性！ 还记得我们在第 10 章学到的踏板焊死公式吗：

Wfinal = W0 + αr (B × A)
训练时所用的 4-bit 底座是经过有损离散化压缩的。如果在残缺的 4-bit 权重上直接做加法，浮点截断误差会被放大数倍，导出的模型会彻底丧失语言逻辑（变成弱智胡说八道）！
因此，Unsloth 会极其严谨地自动从 Hugging Face 官方拉取 Qwen3-4B 的 16-bit 原始全精度权重（约 8GB），在全精度状态下完成踏板物理相加，生成无损的 qwen3-4b.F16.gguf，然后再交给量化器去重新压缩！

  • 产物路径细节排雷：Unsloth 在构建 GGUF 时，底层会自动追加 _gguf 目录标识，最终二进制文件落地为 parenting_qwen3_gguf_gguf/qwen3-4b.Q4_K_M.gguf。通过两级通配符即可将其秒级稳健转存入 Google Drive！
